-- =============================================
-- Draggable Title Bar
-- =============================================
local dragging, dragInput, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)
TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

-- =============================================
-- Helper: สร้างปุ่มสวยงาม
-- =============================================
local function makeButton(parent, text, color, pos, size)
    local btn = Instance.new("TextButton")
    btn.Text = text
    btn.Size = size or UDim2.new(1, 0, 0, 38)
    btn.Position = pos or UDim2.new(0, 0, 0, 0)
    btn.BackgroundColor3 = color or Color3.fromRGB(0, 180, 255)
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextSize = 14
    btn.Font = Enum.Font.GothamSemibold
    btn.BorderSizePixel = 0
    btn.AutoButtonColor = false
    btn.Parent = parent
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 8); c.Parent = btn
    -- Hover
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = color:lerp(Color3.new(1,1,1), 0.15)}):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = color}):Play()
    end)
    return btn
end

local function makeInput(parent, placeholder, pos, size)
    local box = Instance.new("TextBox")
    box.PlaceholderText = placeholder or "พิมพ์ที่นี่..."
    box.Text = ""
    box.Size = size or UDim2.new(1, 0, 0, 36)

    box.BackgroundColor3 = Color3.fromRGB(20, 24, 38)
    box.TextColor3 = Color3.fromRGB(200, 220, 255)
    box.PlaceholderColor3 = Color3.fromRGB(80, 100, 140)
    box.TextSize = 14
    box.Font = Enum.Font.Gotham
    box.BorderSizePixel = 0
    box.ClearTextOnFocus = false
    box.Parent = parent
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 8); c.Parent = box
    local s = Instance.new("UIStroke"); s.Color = Color3.fromRGB(0, 180, 255); s.Thickness = 1.5; s.Transparency = 0.6; s.Parent = box
    local pad = Instance.new("UIPadding"); pad.PaddingLeft = UDim.new(0, 10); pad.Parent = box

-- Notification popup

local function notify(msg, color)
    color = color or Color3.fromRGB(0, 200, 120)
    local notif = Instance.new("Frame")
    notif.Size = UDim2.new(0, 300, 0, 44)
    notif.Position = UDim2.new(0.5, -150, 1, 60)
    notif.BackgroundColor3 = color
    notif.BorderSizePixel = 0
    notif.Parent = ScreenGui
    local nc = Instance.new("UICorner"); nc.CornerRadius = UDim.new(0, 10); nc.Parent = notif
    local nl = Instance.new("TextLabel")
    nl.Text = msg
    nl.Size = UDim2.new(1, -16, 1, 0)
    nl.Position = UDim2.new(0, 8, 0, 0)
    nl.BackgroundTransparency = 1
    nl.TextColor3 = Color3.new(1, 1, 1)
    nl.TextSize = 14
    nl.Font = Enum.Font.GothamSemibold
    nl.Parent = notif

    TweenService:Create(notif, TweenInfo.new(0.3, Enum.EasingStyle.Back), {Position = UDim2.new(0.5, -150, 1, -60)}):Play()
    task.delay(2.5, function()
        TweenService:Create(notif, TweenInfo.new(0.3), {Position = UDim2.new(0.5, -150, 1, 60)}):Play()
        task.delay(0.35, function() notif:Destroy() end)
    end)
end

local function switchTab(idx)
    for i, f in ipairs(tabFrameList) do
        f.Visible = (i == idx)
    end
    for i, t in ipairs(tabs) do
        if i == idx then
            TweenService:Create(t, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(0, 180, 255)}):Play()
            t.TextColor3 = Color3.new(1, 1, 1)
        else
            TweenService:Create(t, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(18, 24, 42)}):Play()
            t.TextColor3 = Color3.fromRGB(120, 160, 210)
        end
    end
    activeTab = idx
end

local guiOpen = false

local function toggleGui()
    guiOpen = not guiOpen
    if guiOpen then
        MainFrame.Visible = true
        MainFrame.Size = UDim2.new(0, 520, 0, 0)
        TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            {Size = UDim2.new(0, 520, 0, 440)}):Play()
        addLog("เปิด Admin Panel")
    else
        TweenService:Create(MainFrame, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            {Size = UDim2.new(0, 520, 0, 0)}):Play()
        task.delay(0.22, function() MainFrame.Visible = false end)
        addLog("ปิด Admin Panel")
    end
end

CloseBtn.MouseButton1Click:Connect(toggleGui)

UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == ADMIN_KEY then
        toggleGui()
    end
end)

-- Neon pulse animation บน border
-- =============================================
task.spawn(function()
    local t = 0
    while true do
        t += 0.05
        local alpha = (math.sin(t) + 1) / 2
        UIStroke.Color = Color3.fromRGB(
            math.floor(0 + alpha * 0),
            math.floor(140 + alpha * 80),
            math.floor(200 + alpha * 55)
        )
        UIStroke.Thickness = 1.5 + alpha * 1.2
        task.wait(0.03)
    end
end)
